const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const cors = require('cors');

const app = express();
app.use(cors());

const server = http.createServer(app);
const io = new Server(server, {
  cors: { origin: "*", methods: ["GET", "POST"] }
});

let gameState = 'WAITING';
let currentMultiplier = 1.00;
let crashPoint = 1.00;
let startTime = null;
let timerSeconds = 5;
let activeBets = [];

function startWaitingPhase() {
  gameState = 'WAITING';
  currentMultiplier = 1.00;
  timerSeconds = 5;
  activeBets = [];

  io.emit('game_state', { state: 'WAITING', timer: timerSeconds });

  const interval = setInterval(() => {
    timerSeconds--;
    io.emit('timer_tick', { timer: timerSeconds });

    if (timerSeconds <= 0) {
      clearInterval(interval);
      startRound();
    }
  }, 1000);
}

function startRound() {
  gameState = 'RUNNING';
  const rand = Math.random();
  crashPoint = rand < 0.08 ? 1.00 : parseFloat((1.01 + Math.pow(rand, 2.5) * 12.0).toFixed(2));
  startTime = Date.now();

  io.emit('game_state', { state: 'RUNNING', crashPoint });

  const gameLoop = setInterval(() => {
    let elapsedTime = (Date.now() - startTime) / 1000;
    currentMultiplier = parseFloat((1.00 + (elapsedTime * 0.3) + Math.pow(elapsedTime, 1.7) * 0.08).toFixed(2));

    if (currentMultiplier >= crashPoint) {
      clearInterval(gameLoop);
      triggerCrash();
    } else {
      io.emit('multiplier_update', { multiplier: currentMultiplier });
    }
  }, 100);
}

function triggerCrash() {
  gameState = 'CRASHED';
  io.emit('game_state', { state: 'CRASHED', crashPoint });

  setTimeout(() => {
    startWaitingPhase();
  }, 3000);
}

io.on('connection', (socket) => {
  socket.emit('init_state', {
    state: gameState,
    multiplier: currentMultiplier,
    bets: activeBets
  });

  socket.on('place_bet', (data) => {
    if (gameState === 'WAITING') {
      const newBet = { id: socket.id, name: data.name, amount: data.amount, cashed: false, win: 0 };
      activeBets.push(newBet);
      io.emit('update_bets', activeBets);
    }
  });

  socket.on('cashout', () => {
    if (gameState === 'RUNNING') {
      const bet = activeBets.find(b => b.id === socket.id);
      if (bet && !bet.cashed) {
        bet.cashed = true;
        bet.win = parseFloat((bet.amount * currentMultiplier).toFixed(3));
        socket.emit('cashout_success', { win: bet.win, multiplier: currentMultiplier });
        io.emit('update_bets', activeBets);
      }
    }
  });
});

startWaitingPhase();

const PORT = process.env.PORT || 3000;
server.listen(PORT, () => {
  console.log(`Servidor Crashy activo en puerto ${PORT}`);
});
